import assert from "node:assert/strict";
import * as fs from "node:fs/promises";
import * as os from "node:os";
import * as path from "node:path";
import { createHash } from "node:crypto";
import { loadHybridFixture, expandedHybridPages } from "./hybrid-retrieval-quality-eval.js";
import { getWikiPageRecords } from "../src/core/retrieval-index.js";
import { retrieveWikiHybrid } from "../src/core/hybrid-retrieval.js";
import { semanticCoverageQueries } from "../src/core/retrieval-coverage.js";
import { PersistentSemanticIndex } from "../src/core/semantic/index.js";
import { LshAnnEngine } from "../src/core/semantic/lsh-engine.js";
import { normalizeVector, storeVector, cosine } from "../src/core/semantic/vector.js";
import { wikiPassageId } from "../src/context/passage-id.js";
import type { EmbeddingProvider } from "../src/core/semantic/types.js";

// A separate untimed audit of saved vectors/results. No live provider or tuning.
const output = path.resolve(process.argv.find((a) => a.startsWith("--output="))?.slice(9) ?? "benchmarks/results/semantic-rescoring-292");
process.env.KNOWLEDGE_RAIL_USAGE_RANKING = "0";
const normalize = (text: string) => text.normalize("NFKC").replace(/\s+/g, " ").trim();
const sha = (value: unknown) => createHash("sha256").update(JSON.stringify(value)).digest("hex");
interface Page { path: string; title: string; type: string; body: string }
interface Query { id: string; query: string; split: string; category: string; relevant: Array<{ path: string; grade: number }>; answerable: boolean; pageTypes?: string[] }
const fixture = JSON.parse(await fs.readFile("benchmarks/fixtures/semantic-rescoring-292.json", "utf8")) as { pages: Page[]; queries: Query[] };
const historical = JSON.parse(await fs.readFile("benchmarks/fixtures/semantic-retrieval-golden.json", "utf8"));
const base = await loadHybridFixture(path.resolve("benchmarks/fixtures", historical.baseFixture));
const corpora = [{ name: "annotated", ...fixture }, { name: "historical", pages: [...expandedHybridPages(base), ...historical.semanticPages] as Page[],
  queries: [...base.queries.map((q) => ({ id: q.id, query: q.query, split: "historical", category: q.case,
    relevant: [...new Map(q.relevant.map((r) => [r.path, { path: r.path, grade: r.grade ?? 1 }])).values()], answerable: q.relevant.length > 0 })),
  ...historical.semanticOnlyQueries.map((q: { id: string; query: string; expectedPaths: string[] }) => ({ ...q, split: "historical", category: "paraphrase", relevant: q.expectedPaths.map((p) => ({ path: p, grade: 2 })), answerable: true }))] as Query[] }];
const stages = ["ann_miss", "ann_threshold", "pool_limit", "rescore_threshold", "page_filter", "duplicate_page", "semantic_selected"] as const;
type Stage = typeof stages[number];
const root = await fs.mkdtemp(path.join(os.tmpdir(), "kr292-losses-"));
try {
  for (const model of ["potion-retrieval-32M", "potion-multilingual-128M", "ollama"]) {
    const reference = JSON.parse(await fs.readFile(path.join(output, `${model}-report.json`), "utf8"));
    const audits = [];
    for (const corpus of corpora) {
      const wiki = path.join(root, model, corpus.name);
      for (const page of corpus.pages) {
        const file = path.join(wiki, page.path); await fs.mkdir(path.dirname(file), { recursive: true });
        await fs.writeFile(file, `---\ntitle: ${JSON.stringify(page.title)}\ntype: ${page.type}\n---\n\n${page.body}`);
      }
      const records = await getWikiPageRecords(wiki, true, { persist: false });
      const passages = records.flatMap((r) => r.passages.map((p) => ({ path: r.path, passageId: wikiPassageId(p), text: `${p.heading}\n${p.text}`.normalize("NFC").trim() })))
        .map((p) => ({ ...p, id: `semantic-${createHash("sha256").update("knowledge-rail-semantic-passage-v1\0").update(p.path).update("\0").update(p.passageId).digest("hex").slice(0, 32)}` }));
      const documents = [...new Set(passages.map((p) => p.text))];
      const texts = [...new Set(corpus.queries.flatMap((q) => [normalize(q.query), ...semanticCoverageQueries(q.query).map((c) => normalize(c.text))]))];
      const frozen = JSON.parse(await fs.readFile(path.join(output, `${model}-${corpus.name}-vectors.json`), "utf8"));
      const expected = reference.reports.find((r: { corpus: string }) => r.corpus === corpus.name);
      assert.equal(sha({ records: records.map((r) => [r.path, r.raw]), queries: corpus.queries }), frozen.corpusDigest);
      assert.equal(sha(frozen), expected.vectorsDigest);
      const docVectors = new Map(documents.map((text, i) => [text, frozen.documents[i] as number[]]));
      const queryVectors = new Map(texts.map((text, i) => [text, frozen.queries[i] as number[]]));
      const get = (map: Map<string, number[]>, key: string) => { const vector = map.get(key); assert.ok(vector); return vector; };
      const provider: EmbeddingProvider = { descriptor: frozen.descriptor,
        async embedDocuments(inputs) { return inputs.map((text) => get(docVectors, text)); },
        async embedQuery(text) { return get(queryVectors, text); },
        async embedQueries(inputs) { return inputs.map((text) => get(queryVectors, text)); },
      };
      const byId = new Map(passages.map((p) => [p.id, p]));
      for (const variant of ["i8-32", "i8-64", "rescore-32", "rescore-64", "f32-32", "f32-64"]) {
        const dtype = variant.startsWith("f32") ? "f32" : "i8";
        const pool = Number(variant.split("-")[1]);
        const variantRoot = path.join(root, model, `${corpus.name}-${variant}`);
        for (const record of records) { const file = path.join(variantRoot, record.path); await fs.mkdir(path.dirname(file), { recursive: true }); await fs.writeFile(file, record.raw); }
        const index = new PersistentSemanticIndex(variantRoot, provider, undefined, { dtype, rescore: variant.startsWith("rescore") });
        // The observer exposes all LSH candidates; production stays at threshold .55.
        const observer = new LshAnnEngine({ dimensions: provider.descriptor.dimensions, minimumScore: -1 });
        observer.rebuild(passages.map((p) => ({ id: p.id, vector: storeVector(get(docVectors, p.text), provider.descriptor.dimensions, dtype).vector, normalized: true })));
        await index.synchronize(records);
        try {
          for (const q of corpus.queries) {
            const vector = get(queryVectors, normalize(q.query));
            const candidates = observer.search(vector, 1000);
            assert.equal(candidates.hits.length, candidates.diagnostics.candidateCount);
            const disposition = new Map<string, Stage>(passages.map((p) => [p.id, "ann_miss"]));
            const eligible = candidates.hits.filter((hit) => hit.score >= 0.55);
            for (const hit of candidates.hits) disposition.set(hit.id, hit.score < 0.55 ? "ann_threshold" : "pool_limit");
            const selected = eligible.slice(0, pool).map((hit) => ({ ...hit }));
            if (variant.startsWith("rescore")) {
              const query = normalizeVector(vector, provider.descriptor.dimensions);
              for (const hit of selected) hit.score = cosine(query, normalizeVector(get(docVectors, byId.get(hit.id)!.text), provider.descriptor.dimensions));
              selected.sort((a, b) => b.score - a.score || a.id.localeCompare(b.id));
            }
            const pagePaths = new Set(records.filter((r) => !q.pageTypes || q.pageTypes.includes(r.type)).map((r) => r.path));
            const seen = new Set<string>();
            const semanticIds = [];
            for (const hit of selected) {
              const passage = byId.get(hit.id)!;
              const stage: Stage = hit.score < 0.55 ? "rescore_threshold" : !pagePaths.has(passage.path) ? "page_filter" : seen.has(passage.path) ? "duplicate_page" : "semantic_selected";
              disposition.set(hit.id, stage);
              if (stage === "semantic_selected") { seen.add(passage.path); semanticIds.push(passage.passageId); }
            }
            const direct = await index.searchWithDiagnostics(q.query, pool, { distinctPages: true, pagePaths });
            assert.deepEqual(direct.hits.map((h) => h.passageId), semanticIds, `${model}/${variant}/${q.id}: stage reconstruction`);
            const counts = Object.fromEntries(stages.map((stage) => [stage, [...disposition.values()].filter((value) => value === stage).length])) as Record<Stage, number>;
            assert.equal(counts.ann_threshold, direct.diagnostics.thresholdRejected);
            assert.equal(counts.pool_limit, direct.diagnostics.poolTruncated);
            assert.equal(counts.rescore_threshold, direct.diagnostics.rescoreThresholdRejected);
            assert.equal(counts.page_filter, direct.diagnostics.filteredPassages);
            assert.equal(counts.duplicate_page, direct.diagnostics.deduplicatedPassages);
            const result = await retrieveWikiHybrid({ wikiRoot: wiki, query: q.query, pageTypes: q.pageTypes,
              maxResults: 8, semanticIndex: index, semanticPoolSize: pool, semanticBudgetMs: 1500,
              initialBudget: { maxEvidence: 8, tokenBudget: 4000 }, maximumBudget: { maxEvidence: 8, tokenBudget: 4000 }, persistDerivedIndexes: false });
            assert.deepEqual(result.hits.map((h) => h.path), expected.outcomes[variant].find((o: { id: string }) => o.id === q.id).paths, `${model}/${variant}/${q.id}: saved final output`);
            const relevant = q.relevant.map((judgment) => {
              const rows = passages.filter((p) => p.path === judgment.path);
              assert.ok(rows.length, `Relevant page absent from corpus: ${judgment.path}`);
              const furthest = Math.max(...rows.map((p) => stages.indexOf(disposition.get(p.id)!)));
              return { ...judgment, semanticStop: stages[furthest], returnedFinal: result.hits.some((h) => h.path === judgment.path),
                passages: rows.map((p) => ({ id: p.passageId, stage: disposition.get(p.id) })) };
            });
            audits.push({ corpus: corpus.name, variant, id: q.id, split: q.split, category: q.category, counts,
              semanticPages: [...seen], finalPaths: result.hits.map((h) => h.path), relevant,
              missedRelevantPages: relevant.filter((r) => !r.returnedFinal).map((r) => ({ path: r.path, stage: r.semanticStop === "semantic_selected" ? "final_selection" : r.semanticStop })) });
          }
        } finally { index.dispose(); observer.dispose(); }
      }
    }
    const summary = [];
    for (const corpus of corpora) for (const variant of ["i8-32", "i8-64", "rescore-32", "rescore-64", "f32-32", "f32-64"]) {
      const rows = audits.filter((a) => a.corpus === corpus.name && a.variant === variant && a.split !== "development");
      const misses = rows.flatMap((r) => r.missedRelevantPages);
      summary.push({ corpus: corpus.name, variant, queries: rows.length, relevant: rows.flatMap((r) => r.relevant).length,
        misses: Object.fromEntries([...stages, "final_selection"].map((stage) => [stage, misses.filter((m) => m.stage === stage).length])),
        passageStages: Object.fromEntries(stages.map((stage) => [stage, rows.reduce((sum, r) => sum + r.counts[stage], 0)])) });
    }
    await fs.writeFile(path.join(output, `${model}-losses.json`), JSON.stringify({ runtimeSourcesSha256: reference.runtimeSourcesSha256, provider: reference.provider,
      method: "Untimed reconstruction from identical frozen vectors, checked against production stage counters, semantic selections, and saved final paths. Threshold -1 is used only by the observer to expose pre-threshold LSH candidates; production remains .55. Page-level relevance does not label every passage relevant. A missing page is attributed to the furthest semantic stage reached by any of its passages; this describes the pipeline, not a counterfactual guarantee that fixing one stage recovers the page. Deduplication preserves one passage per surviving page.", summary, audits }, null, 2));
    console.log(JSON.stringify({ model, assertionsPassed: audits.length, summary: summary.filter((s) => s.corpus === "annotated" && s.variant === "rescore-64") }));
  }
} finally { await fs.rm(root, { recursive: true, force: true }); }
